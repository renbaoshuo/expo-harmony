cmake_minimum_required(VERSION 3.13)
project(expo_network_harmony LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin" AND DEFINED OHOS_SDK_NATIVE)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -include ${OHOS_SDK_NATIVE}/sysroot/usr/include/assert.h")
endif()
if(DEFINED OH_MODULES_DIR)
  set(EXPO_OH_MODULES_DIR "${OH_MODULES_DIR}")
else()
  set(EXPO_OH_MODULES_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../oh_modules")
  set(OH_MODULES_DIR "${EXPO_OH_MODULES_DIR}")
endif()
if(NOT TARGET expo_harmony__expo_modules_core AND NOT TARGET rnoh)
  add_subdirectory("${EXPO_OH_MODULES_DIR}/@expo-harmony/expo-modules-core/src/main/cpp" "${CMAKE_CURRENT_BINARY_DIR}/expo-modules-core")
endif()
add_library(expo_harmony__expo_network SHARED ExpoNetworkProvider.cpp)
target_include_directories(expo_harmony__expo_network PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(expo_harmony__expo_network PUBLIC expo_harmony__expo_modules_core)
